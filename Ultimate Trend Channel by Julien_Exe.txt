mult = 1.0
src = close

// Transform the closing price to a logarithmic scale
src_log = math.log(src)

// Function for calculating upper band using logarithmic scale
upperBandLog(src_log, length) =>
    ta.linreg(src_log, length, 0) + mult * ta.stdev(src_log, length)

// Function for calculating lower band using logarithmic scale
lowerBandLog(src_log, length) =>
    ta.linreg(src_log, length, 0) - mult * ta.stdev(src_log, length)

// Calculation of upper and lower bands for each group using logarithmic scale
upperband1 = upperBandLog(src_log, len_group1)
lowerband1 = lowerBandLog(src_log, len_group1)

// Calculate average of all upper and lower bands on a logarithmic scale
averageBandLog = (upperband1 + lowerband1 + upperband2 + lowerband2 + upperband3 + lowerband3 + upperband4 + lowerband4 + upperband5 + lowerband5 + upperband6 + lowerband6 + upperband7 + lowerband7 + upperband8 + lowerband8 + upperband9 + lowerband9 + upperband10 + lowerband10 + upperband11 + lowerband11 + upperband12 + lowerband12 + upperband13 + lowerband13 + upperband14 + lowerband14 + upperband15 + lowerband15) / 30

// Calculate average of all upper bands on a logarithmic scale
averageUpperBandLog = (upperband1 + upperband2 + upperband3 + upperband4 + upperband5 + upperband6 + upperband7 + upperband8 + upperband9 + upperband10 + upperband11 + upperband12 + upperband13 + upperband14 + upperband15) / 15

// Calculate average of all lower bands on a logarithmic scale
averageLowerBandLog = (lowerband1 + lowerband2 + lowerband3 + lowerband4 + lowerband5 + lowerband6 + lowerband7 + lowerband8 + lowerband9 + lowerband10 + lowerband11 + lowerband12 + lowerband13 + lowerband14 + lowerband15) / 15

// Define fill color for the area between upper and lower bands
fillcolor = color.rgb(0, 191, 255, 95)

// Plot upper and lower bands and fill the area between them
fill(plot(math.exp(averageUpperBandLog), title="Average Upper Band (Log)", color=color.rgb(14, 116, 212), linewidth=1), plot(math.exp(averageLowerBandLog), title="Average Lower Band (Log)", color=color.rgb(14, 116, 212), linewidth=1), color=fillcolor)

// Plot average band
